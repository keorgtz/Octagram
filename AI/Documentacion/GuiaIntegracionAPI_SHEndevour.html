Arquitectura de Validación
Flujo completo del login — online y modo sin conexión
App WPF — Login │ ├─ Generar CodigoInstalacion (MachineGuid → sum estable → 8 chars A–Z) ├─ EmpresaId (config local — usuario ingresa la primera vez, dado por el admin) │ ├─ ¿API disponible? │ │ │ SÍ └─ POST /api/licencia/obtener {EmpresaId, CodigoInstalacion} │ │ │ ├─ 200 OK (licencia encontrada en BD) │ │ ├─ Verificar CodigoInstalacion == hardware actual (seguridad) │ │ ├─ Verificar EmpresaId coincide con el storage (seguridad) │ │ ├─ Verificar FechaLicencia >= hoy (vigencia) │ │ ├─ Actualizar storage local con todos los datos │ │ └─ ✓ PERMITIR acceso │ │ │ └─ 404 Not Found → ✗ BLOQUEAR (equipo sin licencia) │ └─ NO (API caída) → Validar con datos locales ├─ Sin datos locales → ✗ BLOQUEAR ├─ FechaLicencia < hoy → ✗ BLOQUEAR ├─ CodigoInstalacion ≠ equipo → ✗ BLOQUEAR └─ TODO OK → ⚠ PERMITIR (modo sin conexión)

1
Generar el Código de Instalación
Identificador único y estable por equipo — basado en MachineGuid
⚠ CRÍTICO: Este algoritmo debe ser idéntico al del servidor (LicenseGeneratorService). Si cambia tras el primer despliegue, todos los equipos activados fallarán con HardwareInvalido.
C#
using Microsoft.Win32;

public static class MachineHelper
{
    private static string GetMachineGuid()
    {
        using var key = Registry.LocalMachine.OpenSubKey(
            @"SOFTWARE\Microsoft\Cryptography");
        return key?.GetValue("MachineGuid")?.ToString() ?? "UNKNOWN";
    }

    public static string GetCodigoInstalacion()
    {
        string guid = GetMachineGuid();
        int seed = guid.Where(char.IsLetterOrDigit).Sum(c => c);
        var rng = new Random(seed);
        var code = new char[8];
        for (int i = 0; i < 8; i++)
            code[i] = (char)rng.Next(65, 91); // A-Z
        return new string(code);
    }
}
ℹ Se usa .Sum(c => c) en lugar de .GetHashCode() porque GetHashCode() no es estable entre versiones de .NET.
2
Almacenamiento Local — LicenciaLocal
Registro de Windows como storage persistente · EmpresaId y CodigoInstalacion son inmutables
🔒 Regla de seguridad: CodigoInstalacion y EmpresaId se establecen una sola vez y nunca se sobreescriben — evita que un cliente use la licencia de otro.
C#
using Microsoft.Win32;
using System.Text.Json;

public class LicenciaLocal
{
    // Inmutables — primera activación, NUNCA se sobreescriben
    public string CodigoInstalacion { get; set; } = string.Empty;
    public string EmpresaId         { get; set; } = string.Empty; // Identificador permanente del hotel

    // Mutables — se actualizan en cada sincronización exitosa
    public string RFC              { get; set; } = string.Empty;
    public string ClaveLicencia    { get; set; } = string.Empty;
    public string ClaveInstalacion { get; set; } = string.Empty;
    public DateTime FechaLicencia  { get; set; }
    public string NombreComercial  { get; set; } = string.Empty;
    public string RazonSocial      { get; set; } = string.Empty;
    public string NombreCliente    { get; set; } = string.Empty;
    public DateTime UltimaSincronizacion { get; set; }
}

public static class LicenciaStorage
{
    private const string RegPath = @"SOFTWARE\SHEndevour\Licensing";

    public static LicenciaLocal? Leer()
    {
        try
        {
            using var key = Registry.CurrentUser.OpenSubKey(RegPath);
            var json = key?.GetValue("LicenciaData")?.ToString();
            return string.IsNullOrWhiteSpace(json)
                ? null
                : JsonSerializer.Deserialize<LicenciaLocal>(json);
        }
        catch { return null; }
    }

    public static void Guardar(LicenciaLocal nueva, LicenciaLocal? existente)
    {
        var g = existente ?? new LicenciaLocal();

        if (string.IsNullOrWhiteSpace(g.CodigoInstalacion)) g.CodigoInstalacion = nueva.CodigoInstalacion;
        if (string.IsNullOrWhiteSpace(g.EmpresaId))         g.EmpresaId         = nueva.EmpresaId;

        g.RFC                  = nueva.RFC;
        g.ClaveLicencia        = nueva.ClaveLicencia;
        g.ClaveInstalacion     = nueva.ClaveInstalacion;
        g.FechaLicencia        = nueva.FechaLicencia;
        g.NombreComercial      = nueva.NombreComercial;
        g.UltimaSincronizacion = DateTime.UtcNow;

        using var key = Registry.CurrentUser.CreateSubKey(RegPath);
        key.SetValue("LicenciaData", JsonSerializer.Serialize(g));
    }

    public static void Borrar()
    {
        using var key = Registry.CurrentUser.OpenSubKey(RegPath, writable: true);
        key?.DeleteValue("LicenciaData", throwOnMissingValue: false);
    }
}

ℹ La API devuelve los datos tal como están en la BD. La validación criptográfica (verificar que ClaveLicencia corresponde al algoritmo) ocurre en la app WPF con los datos recibidos.
C#
private record ObtenerLicenciaRequest(
    string? EmpresaId,
    string? CodigoInstalacion,
    string? RFC = null);

private record ObtenerLicenciaResponse(
    string EmpresaId, string CodigoInstalacion, string ClaveInstalacion, string ClaveLicencia,
    DateTime FechaLicencia,  string RFC,
    string RazonSocial,      string NombreComercial,  string NombreCliente);

public async Task<ObtenerLicenciaResponse?> ObtenerLicenciaOnlineAsync(
    string empresaId, string codigoInstalacion)
{
    var req = new HttpRequestMessage(HttpMethod.Post, $"{_apiBase}/api/licencia/obtener");
    req.Headers.Add("X-Api-Key", _apiKey);
    req.Content = JsonContent.Create(new ObtenerLicenciaRequest(empresaId, codigoInstalacion));
    try
    {
        var resp = await _http.SendAsync(req);
        if (resp.StatusCode == System.Net.HttpStatusCode.NotFound) return null;
        if (!resp.IsSuccessStatusCode) return null;
        return await resp.Content.ReadFromJsonAsync<ObtenerLicenciaResponse>();
    }
    catch { return null; }
}

Validar sin Conexión (Offline)
Fallback cuando el servidor no está disponible
C#
public static (bool Valida, string Mensaje) ValidarOffline(string codigoMaquina)
{
    var local = LicenciaStorage.Leer();
    if (local == null)
        return (false, "No hay licencia activada en este equipo.");
    if (local.FechaLicencia < DateTime.Today)
        return (false, $"La licencia expiró el {local.FechaLicencia:dd/MM/yyyy}.");
    if (local.CodigoInstalacion != codigoMaquina)
        return (false, "La licencia almacenada no corresponde a este equipo.");
    return (true, string.Empty);
}
⚠ El modo offline solo funciona si la licencia fue validada online al menos una vez. Sin datos en el storage, se bloquea el acceso.

Flujo Completo en el Login
Método principal — integra online + offline + verificaciones de seguridad
C#
public async Task<bool> VerificarLicenciaAlLoginAsync()
{
    string codigoMaquina = MachineHelper.GetCodigoInstalacion();
    LicenciaLocal? localActual = LicenciaStorage.Leer();

    // Primera vez: pedir EmpresaId al usuario (el admin se lo proporciona)
    string? empresaId = localActual?.EmpresaId;
    if (string.IsNullOrWhiteSpace(empresaId))
    {
        empresaId = MostrarDialogoEmpresaId(codigoMaquina); // muestra CodigoInstalacion, pide EmpresaId
        if (string.IsNullOrWhiteSpace(empresaId)) return false;
    }

    if (await TestConnectionAsync())
    {
        var r = await ObtenerLicenciaOnlineAsync(empresaId, codigoMaquina);

        if (r == null)
        {
            MostrarError("No existe licencia activa para este equipo. " +
                         $"Proporciona al administrador el código: {codigoMaquina}");
            return false;
        }

        // Verificaciones de seguridad
        if (r.CodigoInstalacion != codigoMaquina)
        { MostrarError("Inconsistencia de hardware. Contacta a tu proveedor."); return false; }

        if (!string.Equals(r.EmpresaId, empresaId, StringComparison.OrdinalIgnoreCase))
        { MostrarError("El EmpresaId no coincide con la licencia registrada."); return false; }

        if (r.FechaLicencia < DateTime.Today)
        { MostrarError($"La licencia venció el {r.FechaLicencia:dd/MM/yyyy}. Renuévala con tu proveedor."); return false; }

        LicenciaStorage.Guardar(new LicenciaLocal
        {
            EmpresaId         = r.EmpresaId,
            CodigoInstalacion = r.CodigoInstalacion,
            RFC               = r.RFC,
            ClaveLicencia     = r.ClaveLicencia,
            ClaveInstalacion  = r.ClaveInstalacion,
            FechaLicencia     = r.FechaLicencia,
            NombreComercial   = r.NombreComercial,
            RazonSocial       = r.RazonSocial,
            NombreCliente     = r.NombreCliente
        }, localActual);

        if (r.FechaLicencia <= DateTime.Today.AddDays(30))
            MessageBox.Show($"Tu licencia vence el {r.FechaLicencia:dd/MM/yyyy}.",
                "Aviso", MessageBoxButton.OK, MessageBoxImage.Information);
        return true;
    }

    var (valida, msg2) = ValidarOffline(codigoMaquina);
    if (!valida) { MostrarError("Sin conexión.\n" + msg2); return false; }
    MessageBox.Show(
        $"Modo sin conexión. Licencia válida hasta {localActual!.FechaLicencia:dd/MM/yyyy}.",
        "Modo sin conexión", MessageBoxButton.OK, MessageBoxImage.Information);
    return true;
}

Secretos y Seguridad

Qué son, cómo funcionan y cómo generar API Keys, JWT Secret y contraseñas seguras

🔐
API Key — Clave compartida cliente↔servidor
Protege todos los endpoints de negocio de la API
API_KEY
¿Qué es?
Una cadena aleatoria larga que identifica a tu aplicación WPF frente a la API. No identifica a un usuario, identifica al cliente de la API. Es como una contraseña compartida entre la app y el servidor.
App WPF → POST /api/licencia/validar
Header: X-Api-Key: TU_API_KEY
API → Compara con ApiKey en appsettings / .env
Si no coincide → 401 Unauthorized
⚠
Nunca expongas la API Key en:
· Código fuente público en GitHub
· Variables de entorno sin cifrar en CI/CD sin protección
· Mensajes de error o logs del servidor
💡 En el cliente WPF es aceptable tenerla en App.config cifrado o como recurso embebido, ya que el usuario final no debería tener acceso al binario descompilado. Para extra seguridad, usa ofuscación.
🎫
JWT Secret — Firma de tokens del panel admin
Autentica a los usuarios del panel web de licencias
JWT_SECRET
¿Qué es un JWT?
JSON Web Token — un ticket firmado digitalmente. Cuando un administrador hace login en el panel web, la API genera un JWT firmado con el JWT Secret. En cada request subsiguiente, el panel manda ese token y la API lo verifica con el mismo secret.
Admin login → POST /api/auth/login (user+pass)
API → Genera JWT firmado con JWT_SECRET
Panel Web → Almacena JWT · lo envía en cada request
API → Verifica firma con JWT_SECRET
Si ok → acceso permitido | si no → 401
⚠
Si el JWT Secret se filtra:
· Un atacante puede forjar tokens admin válidos sin conocer ninguna contraseña
· Solución: cambiar el secret inmediatamente (invalida TODOS los tokens activos)
· Mínimo 32 caracteres aleatorios, idealmente 48+
ℹ El JWT tiene un tiempo de expiración (ExpirationMinutes: 480 = 8 horas). Después el admin debe hacer login de nuevo. No afecta a los clientes WPF — ellos usan API Key, no JWT.
🗄
SQL Server — Contraseña de Base de Datos
Usuario sa · configuración actual del proyecto
SQL SA
Usuario sa (System Administrator)
El usuario sa es el administrador total de SQL Server. El proyecto está configurado para usar sa / InfoHotel01 tanto en Windows como en Docker. Esto simplifica la configuración pero da acceso completo a todas las bases de datos del servidor.
docker-compose.yml → SA_PASSWORD: "InfoHotel01"
Contenedor API/Web → User Id=sa;Password=InfoHotel01
SQL Server → Acceso completo a SHEWebLicenseDb
⚠ En producción pública se recomienda un usuario dedicado con permisos mínimos. Para uso interno/pyme con SQL Server en Docker aislado, sa es aceptable.
⚙
Cómo Generar Valores Seguros
Comandos para generar API Key, JWT Secret y contraseñas
✓ Regla de oro: genera secretos con herramientas criptográficas, nunca a mano ni con patrones predecibles. Usa una de estas opciones:

PowerShell (Windows)

PowerShell
# JWT Secret (64 chars base64)
[Convert]::ToBase64String(
  (1..48 | % { Get-Random -Max 256 }))

# API Key (hex, 64 chars)
-join ((1..32) | % {
  '{0:x2}' -f (Get-Random -Max 256) })

Bash (Linux / Ubuntu)

Bash
# JWT Secret (base64, 64 chars)
openssl rand -base64 48

# API Key (hex, 64 chars)
openssl rand -hex 32
📋
Resumen de secretos del proyecto:
JWT_SECRET → mínimo 32 chars · genera con openssl · expira tokens del panel admin
API_KEY → hex de 32 bytes · va en el header X-Api-Key de la app WPF
SA_PASSWORD → mínimo 8 chars · mayúscula + minúscula + número (req. SQL Server)

🗺
Arquitectura de red en el servidor
Cómo fluye el tráfico desde internet hasta los contenedores
Internet ↓ Cloudflare DNS (CNAME → TUNNEL_ID.cfargotunnel.com · Proxy ON) ↓ Cloudflare Tunnel (cifrado · sin abrir puertos en el router) ↓ cloudflared (contenedor Docker · red proxy · apunta todo a npm:80) ↓ Nginx Proxy Manager (contenedor Docker · red proxy · puerto 80/443) ↓ (enruta por hostname) ├─ api.endevour.mx → she-license-api:5000 └─ panel.endevour.mx → she-license-web:5287 ↓ she-sqlserver:1433 (solo red interna, sin expose público)
ℹ Todos los contenedores comparten la red Docker proxy. Los de SHEWeb usan expose (no ports) — ningún puerto queda accesible desde la IP del servidor. Solo NPM puede alcanzarlos, por nombre de contenedor dentro de la red.

WebSockets en Program.cs
Tres cambios obligatorios — sin ellos Blazor Server no funciona en producción
ⓘ Este código ya está aplicado en el repositorio. Revisa esta sección para entender por qué cada cambio es necesario y cómo verificarlo si algo no funciona.
C# — SHE.License.Web/Program.cs (partes críticas para WebSocket)
// ① Circuit retention: retiene el circuit 10 min tras desconexión
//    (tab minimizado, red caída) — evita reiniciar sesión al reconectar
builder.Services.AddRazorComponents()
    .AddInteractiveServerComponents(options =>
    {
        options.DisconnectedCircuitRetentionPeriod = TimeSpan.FromMinutes(10);
    });

// ② SignalR keepalive: Cloudflare cierra conexiones idle a 100 s
//    KeepAliveInterval = 15 s → ping cada 15 s → nunca alcanza el timeout
builder.Services.AddSignalR(options =>
{
    options.ClientTimeoutInterval     = TimeSpan.FromSeconds(30);
    options.HandshakeTimeout          = TimeSpan.FromSeconds(15);
    options.KeepAliveInterval         = TimeSpan.FromSeconds(15);
    options.MaximumReceiveMessageSize = 32 * 1024;
});

// ... (DB, repositorios, auth, MediatR — sin cambios WebSocket) ...

var app = builder.Build();

// En producción: NPM y Cloudflare manejan TLS; el contenedor solo recibe HTTP.
// UseHttpsRedirection causaría bucle HTTP→HTTPS dentro del contenedor.
if (app.Environment.IsDevelopment())
{
    app.UseHttpsRedirection();
}
else
{
    app.UseExceptionHandler("/Error", createScopeForErrors: true);
}

// ③ UseWebSockets ANTES de UseAuthentication y del routing de componentes.
//    Kestrel intercepta el handshake HTTP→WS antes de que el router procese la petición.
app.UseWebSockets();
app.UseAuthentication();
app.UseAntiforgery();

app.MapStaticAssets();
app.MapRazorComponents<App>()
    .AddInteractiveServerRenderMode();

app.Run();
CambioSin este cambioMotivo
DisconnectedCircuitRetentionPeriod = 10 min El estado del circuit se destruye en segundos tras desconexión; al reconectar el usuario pierde todo el estado de la UI Mantiene en memoria el circuit de cada tab durante 10 min; la reconexiones automáticas de Blazor recuperan el estado sin recargar
KeepAliveInterval = 15 s Cloudflare cierra la conexión WebSocket a los 100 s de inactividad; el panel se desconecta silenciosamente mientras el usuario lee SignalR envía un ping cada 15 s; la conexión nunca supera el umbral de 100 s de Cloudflare
app.UseWebSockets() antes de UseAuthentication El middleware de auth o antiforgery ve la petición WebSocket como HTTP normal y la rechaza con 400 o 401 Kestrel procesa el handshake Upgrade: websocket antes de que el pipeline de middlewares lo intercepte
UseHttpsRedirection solo en Development En producción el contenedor recibe HTTP de NPM; redirigir a HTTPS genera un bucle de redirección infinita (HTTP 307 → HTTP 307 → ...) NPM y Cloudflare ya terminan TLS; el contenedor solo necesita HTTP internamente

Script de backup

ℹ Crea /root/backup_sheweb.sh y dale permisos de ejecución una sola vez.
Bash
#!/bin/bash
set -e
DATE=$(date +%Y%m%d_%H%M%S)
BACKUP_DIR="/root/backups/$DATE"
APP_DIR="/root/docker/apps/SHEWeb"

mkdir -p "$BACKUP_DIR"

echo "[1/4] Base de datos..."
docker exec she-sqlserver /opt/mssql-tools18/bin/sqlcmd \
    -S localhost -U sa -P "InfoHotel01" -C -N \
    -Q "BACKUP DATABASE SHEWebLicenseDb TO DISK='/var/opt/mssql/bak_tmp.bak' WITH FORMAT, COMPRESSION"
docker cp she-sqlserver:/var/opt/mssql/bak_tmp.bak "$BACKUP_DIR/db.bak"
docker exec she-sqlserver rm /var/opt/mssql/bak_tmp.bak

echo "[2/4] Secretos y configuración..."
cp "$APP_DIR/.env"              "$BACKUP_DIR/.env"
cp "$APP_DIR/docker-compose.yml" "$BACKUP_DIR/docker-compose.yml"

echo "[3/4] Cloudflare tunnel..."
cp -r /root/docker/cloudflared/ "$BACKUP_DIR/cloudflared/"

echo "[4/4] Nginx Proxy Manager..."
cp -r /root/docker/npm_data/ "$BACKUP_DIR/npm_data/" 2>/dev/null \
    || echo "  (npm_data no encontrado — omitido)"

tar -czf "/root/backups/sheweb_${DATE}.tar.gz" -C "/root/backups" "$DATE"
rm -rf "$BACKUP_DIR"
echo "✓ Backup: /root/backups/sheweb_${DATE}.tar.gz"

Transferir backups fuera del servidor

Bash
# Opción A — SCP a otra máquina
scp /root/backups/sheweb_*.tar.gz usuario@OTRO_SERVIDOR:/backups/

# Opción B — rclone a Google Drive (recomendado)
curl https://rclone.org/install.sh | sudo bash
rclone config                  # sigue el asistente, elige Google Drive
rclone copy /root/backups/ gdrive:Backups/SHEWeb/ --include "*.tar.gz"

# Agregar rclone al cron (30 min después del backup)
30 2 * * * rclone copy /root/backups/ gdrive:Backups/SHEWeb/ --include "*.tar.gz" >> /root/backups/backup.log 2>&1

# Limpiar backups locales con más de 7 días
find /root/backups/ -name "sheweb_*.tar.gz" -mtime +7 -delete

Restauración en servidor nuevo

Bash
# ── PREREQUISITOS ────────────────────────────────────────────────────────────
# Servidor nuevo con Docker + Docker Compose instalados (GuiaUbuntuServer.md)
docker network create proxy

# ── EXTRAER BACKUP ────────────────────────────────────────────────────────────
scp usuario@ORIGEN:/root/backups/sheweb_FECHA.tar.gz /tmp/
cd /tmp && tar -xzf sheweb_FECHA.tar.gz
BACKUP="/tmp/FECHA"       # ajusta al nombre de carpeta extraída

# ── CLOUDFLARED ───────────────────────────────────────────────────────────────
mkdir -p /root/docker/cloudflared/
cp -r $BACKUP/cloudflared/* /root/docker/cloudflared/
# Levantar cloudflared con su docker-compose

# ── NPM ───────────────────────────────────────────────────────────────────────
mkdir -p /root/docker/npm_data/
cp -r $BACKUP/npm_data/* /root/docker/npm_data/
# Levantar NPM — proxy hosts y certificados ya estarán presentes

# ── APLICACIÓN ────────────────────────────────────────────────────────────────
git clone https://github.com/TU_USUARIO/SHEWeb.git /root/docker/apps/SHEWeb
cp "$BACKUP/.env" /root/docker/apps/SHEWeb/.env

# ── CONTENEDORES ──────────────────────────────────────────────────────────────
cd /root/docker/apps/SHEWeb
docker compose up -d
# Esperar ~60 s a que SQL Server esté healthy
watch docker compose ps    # Ctrl+C cuando los tres estén Running

# ── RESTAURAR BD ──────────────────────────────────────────────────────────────
docker cp "$BACKUP/db.bak" she-sqlserver:/var/opt/mssql/restore.bak

docker exec she-sqlserver /opt/mssql-tools18/bin/sqlcmd \
    -S localhost -U sa -P "InfoHotel01" -C -N \
    -Q "IF NOT EXISTS (SELECT name FROM sys.databases WHERE name='SHEWebLicenseDb') \
        CREATE DATABASE SHEWebLicenseDb;"

docker exec she-sqlserver /opt/mssql-tools18/bin/sqlcmd \
    -S localhost -U sa -P "InfoHotel01" -C -N \
    -Q "RESTORE DATABASE SHEWebLicenseDb FROM DISK='/var/opt/mssql/restore.bak' \
        WITH REPLACE, RECOVERY;"

docker exec she-sqlserver rm /var/opt/mssql/restore.bak

# ── VERIFICAR ─────────────────────────────────────────────────────────────────
docker exec she-license-api curl http://localhost:5000/api/health
